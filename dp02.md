# 62. Unique Paths
```java
class Solution {
    public int uniquePaths(int m, int n) {
        // The problem asks: given an m x n grid, how many unique paths exist 
        // from the top-left corner to the bottom-right corner 
        // if the robot can only move right or down at any step.

        // --- Step 1: Define a DP table ---
        // dp[i][j] will represent the number of unique paths 
        // to reach cell (i, j) from the start (0, 0).
        int[][] dp = new int[m][n];

        // --- Step 2: Initialize the starting position ---
        // There's exactly one way to be at the starting cell (0, 0) — just start there.
        dp[0][0] = 1;

        // --- Step 3: Initialize the first row ---
        // The robot can only move right along the first row.
        // Therefore, every cell in the first row can be reached in exactly one way.
        for (int j = 1; j < n; j++) {
            dp[0][j] = 1;
        }

        // --- Step 4: Initialize the first column ---
        // Similarly, the robot can only move down along the first column.
        // So each cell in the first column can be reached in exactly one way.
        for (int i = 1; i < m; i++) {
            dp[i][0] = 1;
        }

        // --- Step 5: Fill the rest of the DP table ---
        // For each cell (i, j), the robot could have come from either:
        //   - the cell above it (i-1, j)
        //   - or the cell to the left (i, j-1)
        // So the total number of unique paths to (i, j)
        // is the sum of paths from those two directions.
        for (int i = 1; i < m; i++) {
            for (int j = 1; j < n; j++) {
                dp[i][j] = dp[i-1][j] + dp[i][j-1];
            }
        }

        // --- Step 6: Return the result ---
        // The bottom-right cell (m-1, n-1) now contains 
        // the total number of unique paths.
        return dp[m-1][n-1];
    }
}
```

# 63. Unique Paths II
```java
class Solution {
    public int uniquePathsWithObstacles(int[][] obstacleGrid) {
        // Step 1: Get dimensions of the grid
        int row = obstacleGrid.length;
        int column = obstacleGrid[0].length;

        // Step 2: If the starting cell itself has an obstacle, there are 0 valid paths
        if (obstacleGrid[0][0] == 1) {
            return 0;
        }

        // Step 3: Initialize the starting point
        // Using 1 to represent one valid way to reach the starting cell
        obstacleGrid[0][0] = 1;

        // Step 4: Fill the first row
        // Since we can only move right in the first row, 
        // a cell is reachable if all previous cells to its left are reachable and it has no obstacle
        for (int j = 1; j < column; j++) {
            // If current cell has no obstacle and the cell to the left is reachable
            if (obstacleGrid[0][j] == 0 && obstacleGrid[0][j - 1] == 1) {
                obstacleGrid[0][j] = 1; // Mark as reachable
            } else {
                obstacleGrid[0][j] = 0; // Otherwise, unreachable
            }
        }

        // Step 5: Fill the first column
        // Similarly, we can only move down in the first column.
        for (int i = 1; i < row; i++) {
            // If current cell has no obstacle and the cell above it is reachable
            if (obstacleGrid[i][0] == 0 && obstacleGrid[i - 1][0] == 1) {
                obstacleGrid[i][0] = 1; // Mark as reachable
            } else {
                obstacleGrid[i][0] = 0; // Otherwise, unreachable
            }
        }

        // Step 6: Fill the rest of the grid using Dynamic Programming
        // Transition formula:
        // If the current cell has no obstacle:
        //   paths[i][j] = paths[i - 1][j] (from top) + paths[i][j - 1] (from left)
        // Otherwise:
        //   paths[i][j] = 0 (unreachable)
        for (int i = 1; i < row; i++) {
            for (int j = 1; j < column; j++) {
                if (obstacleGrid[i][j] == 0) {
                    int up = obstacleGrid[i - 1][j];     // Paths from the cell above
                    int left = obstacleGrid[i][j - 1];   // Paths from the cell to the left
                    obstacleGrid[i][j] = up + left;      // Total ways to reach this cell
                } else {
                    obstacleGrid[i][j] = 0;              // Obstacle cell — no paths
                }
            }
        }

        // Step 7: Return the number of ways to reach the bottom-right corner
        return obstacleGrid[row - 1][column - 1];
    }
}
```
